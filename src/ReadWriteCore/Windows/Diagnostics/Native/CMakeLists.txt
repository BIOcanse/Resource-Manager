cmake_minimum_required(VERSION 3.24)
project(ResourceManagerDumpAnalysis LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT (MSVC OR MINGW))
  message(FATAL_ERROR "The dump helper requires Windows x64 and platform DbgEng headers (MSVC or MinGW-w64)")
endif()
enable_language(RC)
get_filename_component(PRODUCT_ICON_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../Shared/Assets/ProductIcon" ABSOLUTE)
add_executable(ResourceManager.DumpAnalysis WindowsDumpAnalysis.cpp "${PRODUCT_ICON_DIR}/product.rc")
target_include_directories(ResourceManager.DumpAnalysis PRIVATE "${PRODUCT_ICON_DIR}")
target_compile_features(ResourceManager.DumpAnalysis PRIVATE cxx_std_20)
target_compile_definitions(ResourceManager.DumpAnalysis PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
if(MSVC)
  target_compile_options(ResourceManager.DumpAnalysis PRIVATE /W4 /WX /permissive- /utf-8)
  set_property(TARGET ResourceManager.DumpAnalysis PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  target_compile_options(ResourceManager.DumpAnalysis PRIVATE -Wall -Wextra -Werror)
  target_link_options(ResourceManager.DumpAnalysis PRIVATE -municode -static)
endif()
target_link_libraries(ResourceManager.DumpAnalysis PRIVATE ole32)
# dbgeng.dll is loaded only from the explicit supplied installed-engine path.
